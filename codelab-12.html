Crear un fake de Repository

Crea test/fake_task_repository.dart. Ajusta las firmas si ampliaste el contrato en la sesión 10.

import 'package:taskflow/features/tasks/data/task_repository.dart';
import 'package:taskflow/features/tasks/domain/task.dart';

class FakeTaskRepository implements TaskRepository {
  FakeTaskRepository(this.items);
  final List<Task> items;
  @override
  Future<List<Task>> getTasks() async => List.of(items);
  @override
  void save(Task task) {
    final index = items.indexWhere((item) => item.id == task.id);
    if (index >= 0) items[index] = task;
  }
  @override
  Future<Task> createTask(String title) async {
    final task = Task(id: items.length + 1, title: title,
      description: '', priority: 'Media', isCompleted: false, userId: 1);
    items.add(task);
    return task;
  }
  @override
  Future<void> updateTask(Task task) async => save(task);
  @override
  Future<void> deleteTask(int id) async => items.removeWhere((t) => t.id == id);
}

Un fake permite provocar éxito, vacío o error de manera determinista.

Probar formulario y pantalla vacía

Crea test/task_widgets_test.dart y pega el archivo completo. La primera prueba comprueba que Guardar no acepta un título vacío. La segunda usa el fake para comprobar el estado vacío sin hacer red.

import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:provider/provider.dart';
import 'package:taskflow/features/tasks/presentation/task_form_screen.dart';
import 'package:taskflow/features/tasks/presentation/task_list_screen.dart';
import 'package:taskflow/features/tasks/presentation/tasks_view_model.dart';
import 'fake_task_repository.dart';

void main() {
  testWidgets('formulario pide un título', (tester) async {
    await tester.pumpWidget(const MaterialApp(home: TaskFormScreen()));
    await tester.tap(find.text('Guardar'));
    await tester.pump();
    expect(find.text('Escribe un título.'), findsOneWidget);
  });

  testWidgets('lista muestra estado vacío', (tester) async {
    final model = TasksViewModel(FakeTaskRepository([]));
    await model.loadTasks();
    await tester.pumpWidget(ChangeNotifierProvider<TasksViewModel>.value(
      value: model,
      child: const MaterialApp(home: TaskListScreen()),
    ));
    expect(find.text('Todavía no hay tareas para mostrar.'), findsOneWidget);
  });
}
